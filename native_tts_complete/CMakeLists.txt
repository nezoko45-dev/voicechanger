cmake_minimum_required(VERSION 3.20)
project(native_tts_complete LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(whispercpp GIT_REPOSITORY https://github.com/ggml-org/whisper.cpp.git GIT_TAG master GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(whispercpp)
add_library(whisper SHARED whisper_dll.cpp)
target_include_directories(whisper PRIVATE ${whispercpp_SOURCE_DIR}/include)
target_link_libraries(whisper PRIVATE whisper.lib)
set_target_properties(whisper PROPERTIES OUTPUT_NAME whisper)
add_library(tts SHARED tts_dll.cpp)
target_include_directories(tts PRIVATE .)
set_target_properties(tts PROPERTIES OUTPUT_NAME tts)
add_executable(tts_app host.cpp)
target_include_directories(tts_app PRIVATE . ${whispercpp_SOURCE_DIR}/include)
target_link_libraries(tts_app PRIVATE winmm whisper tts)
